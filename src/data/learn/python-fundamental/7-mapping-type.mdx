# Python Mapping Types - Complete Guide

## Overview

Mapping types in Python represent collections of key-value pairs. The primary built-in mapping type is `dict` (dictionary), which implements a hash table data structure. Python also provides specialized mapping types in the `collections` module.

---

## 1. Dictionary (dict) - Hash Table Implementation

### Description
Dictionaries are mutable mappings that store key-value pairs using a hash table implementation. Since Python 3.7, dictionaries maintain insertion order as part of the language specification. Keys must be hashable (immutable and have a `__hash__()` method).

### Constructor
```python
# Various ways to create dictionaries
dict()                          # {} (empty dictionary)
dict([('a', 1), ('b', 2)])     # From sequence of key-value pairs
dict(a=1, b=2, c=3)            # Using keyword arguments
dict(zip(['a', 'b'], [1, 2]))  # From two sequences
dict({'a': 1, 'b': 2})         # From another dictionary (copy)
{'a': 1, 'b': 2}               # Dictionary literal
{x: x**2 for x in range(5)}    # Dictionary comprehension
```

### Key Attributes
- **No specific attributes** - dictionaries are mutable mapping objects

### Methods - Accessing and Modifying
```python
d = {'a': 1, 'b': 2, 'c': 3}

# Accessing values
d['a']                  # 1 - direct access (raises KeyError if not found)
d.get('a')              # 1 - safe access
d.get('d', 'default')   # 'default' - with default value
d.setdefault('d', 4)    # 4 - get value or set and return default

# Modifying values
d['a'] = 10             # Set value
d.update({'d': 4, 'e': 5})      # Update with another dict
d.update([('f', 6), ('g', 7)])  # Update with sequence of pairs
d.update(h=8, i=9)              # Update with keyword arguments

# Removing items
value = d.pop('a')              # 10 - remove and return value
value = d.pop('z', 'not found') # 'not found' - with default
key, value = d.popitem()        # Remove and return arbitrary item
del d['b']                      # Remove specific key
d.clear()                       # Remove all items
```

### Methods - Dictionary Views
```python
d = {'a': 1, 'b': 2, 'c': 3}

# Get views (these are dynamic and reflect changes)
keys = d.keys()         # dict_keys(['a', 'b', 'c'])
values = d.values()     # dict_values([1, 2, 3])
items = d.items()       # dict_items([('a', 1), ('b', 2), ('c', 3)])

# Views are dynamic
d['d'] = 4
print(list(keys))       # ['a', 'b', 'c', 'd'] - automatically updated

# View operations
print('a' in keys)      # True - membership testing
print(len(keys))        # 4 - length

# Set operations on views
d1 = {'a': 1, 'b': 2}
d2 = {'b': 2, 'c': 3}
print(d1.keys() & d2.keys())    # {'b'} - intersection
print(d1.keys() | d2.keys())    # {'a', 'b', 'c'} - union
print(d1.keys() - d2.keys())    # {'a'} - difference
```

### Dictionary Operations and Methods
```python
# Copying dictionaries
original = {'a': 1, 'b': [2, 3]}
shallow_copy = original.copy()          # Shallow copy
import copy
deep_copy = copy.deepcopy(original)     # Deep copy

# Dictionary comparison
d1 = {'a': 1, 'b': 2}
d2 = {'b': 2, 'a': 1}
print(d1 == d2)         # True - order doesn't matter for equality

# Merging dictionaries (Python 3.9+)
dict1 = {'a': 1, 'b': 2}
dict2 = {'c': 3, 'd': 4}
dict3 = {'b': 20, 'e': 5}

merged = dict1 | dict2              # {'a': 1, 'b': 2, 'c': 3, 'd': 4}
merged = dict1 | dict2 | dict3      # {'a': 1, 'b': 20, 'c': 3, 'd': 4, 'e': 5}

# In-place merge
dict1 |= dict2          # Modifies dict1

# Pre-Python 3.9 merging
merged = {**dict1, **dict2}         # Using unpacking
merged = dict(dict1, **dict2)       # Using dict constructor
```

### Dictionary Comprehensions
```python
# Basic dictionary comprehension
squares = {x: x**2 for x in range(5)}  # {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# With condition
even_squares = {x: x**2 for x in range(10) if x % 2 == 0}

# From existing dictionary
original = {'a': 1, 'b': 2, 'c': 3, 'd': 4}
filtered = {k: v for k, v in original.items() if v > 2}    # {'c': 3, 'd': 4}
transformed = {k: v*2 for k, v in original.items()}        # {'a': 2, 'b': 4, 'c': 6, 'd': 8}

# Swapping keys and values
swapped = {v: k for k, v in original.items()}              # {1: 'a', 2: 'b', 3: 'c', 4: 'd'}

# Creating from two lists
keys = ['name', 'age', 'city']
values = ['Alice', 30, 'New York']
person = {k: v for k, v in zip(keys, values)}
```

### Hash Table Concepts
```python
# Understanding hashability
hashable_keys = [
    42,                 # int
    3.14,              # float
    "hello",           # str
    (1, 2, 3),         # tuple (if all elements are hashable)
    frozenset([1, 2])  # frozenset
]

# Non-hashable keys (will raise TypeError)
# non_hashable = {
#     [1, 2]: "list",      # list is not hashable
#     {1, 2}: "set",       # set is not hashable
#     {'a': 1}: "dict"     # dict is not hashable
# }

# Custom hash function
class Point:
    def __init__(self, x, y):
        self.x, self.y = x, y
    
    def __hash__(self):
        return hash((self.x, self.y))
    
    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

# Now Point objects can be dictionary keys
points = {Point(1, 2): "origin", Point(3, 4): "point1"}
```

### Advanced Dictionary Patterns
```python
# Counting with dictionaries
text = "hello world"
char_count = {}
for char in text:
    char_count[char] = char_count.get(char, 0) + 1

# Or using setdefault
char_count = {}
for char in text:
    char_count.setdefault(char, 0)
    char_count[char] += 1

# Grouping with dictionaries
students = [
    {'name': 'Alice', 'grade': 'A', 'subject': 'Math'},
    {'name': 'Bob', 'grade': 'B', 'subject': 'Math'},
    {'name': 'Charlie', 'grade': 'A', 'subject': 'Science'},
]

# Group by grade
by_grade = {}
for student in students:
    grade = student['grade']
    by_grade.setdefault(grade, []).append(student)

# Nested dictionaries
nested = {
    'user1': {'name': 'Alice', 'settings': {'theme': 'dark', 'lang': 'en'}},
    'user2': {'name': 'Bob', 'settings': {'theme': 'light', 'lang': 'es'}}
}

# Safe nested access
def safe_get(d, *keys):
    for key in keys:
        if isinstance(d, dict) and key in d:
            d = d[key]
        else:
            return None
    return d

theme = safe_get(nested, 'user1', 'settings', 'theme')  # 'dark'
```

## 2. Specialized Mapping Types (collections module)

### 2.1 defaultdict - Dictionary with Default Factory

```python
from collections import defaultdict

# Constructor
dd = defaultdict(list)          # Default factory: list
dd = defaultdict(int)           # Default factory: int (returns 0)
dd = defaultdict(set)           # Default factory: set
dd = defaultdict(lambda: 'N/A') # Custom default factory

# Usage
dd = defaultdict(list)
dd['fruits'].append('apple')    # Automatically creates empty list
dd['fruits'].append('banana')
dd['vegetables'].append('carrot')

print(dict(dd))  # {'fruits': ['apple', 'banana'], 'vegetables': ['carrot']}

# Counting example
from collections import defaultdict
text = "hello world"
char_count = defaultdict(int)
for char in text:
    char_count[char] += 1       # No need to check if key exists

# Grouping example
students = [('Alice', 'A'), ('Bob', 'B'), ('Charlie', 'A')]
by_grade = defaultdict(list)
for name, grade in students:
    by_grade[grade].append(name)

# Access default_factory
print(dd.default_factory)       # <class 'list'>
dd.default_factory = None       # Disable default factory
```

### 2.2 OrderedDict - Order-Preserving Dictionary

```python
from collections import OrderedDict

# Constructor (less relevant since Python 3.7+ dicts maintain order)
od = OrderedDict()
od = OrderedDict([('a', 1), ('b', 2)])
od = OrderedDict(a=1, b=2)

# Special methods
od = OrderedDict([('a', 1), ('b', 2), ('c', 3)])
od.move_to_end('a')             # Move 'a' to end
od.move_to_end('c', last=False) # Move 'c' to beginning

# Pop from specific end
last_item = od.popitem(last=True)   # Remove from end
first_item = od.popitem(last=False) # Remove from beginning

# Use case: LRU Cache implementation
class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()
    
    def get(self, key):
        if key in self.cache:
            # Move to end (most recently used)
            self.cache.move_to_end(key)
            return self.cache[key]
        return -1
    
    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        elif len(self.cache) >= self.capacity:
            # Remove least recently used (first item)
            self.cache.popitem(last=False)
        self.cache[key] = value
```

### 2.3 Counter - Counting Hashable Objects

```python
from collections import Counter

# Constructor
c = Counter()                           # Empty counter
c = Counter("hello world")              # From string
c = Counter(['a', 'b', 'c', 'a'])      # From list
c = Counter({'a': 3, 'b': 1})          # From dict
c = Counter(a=3, b=1)                   # From keyword args

# Most common elements
text = "hello world"
char_count = Counter(text)
print(char_count.most_common())         # [('l', 3), ('o', 2), ('h', 1), ...]
print(char_count.most_common(3))        # Top 3: [('l', 3), ('o', 2), ('h', 1)]

# Counter arithmetic
c1 = Counter(['a', 'b', 'c', 'a'])      # Counter({'a': 2, 'b': 1, 'c': 1})
c2 = Counter(['a', 'b', 'b'])           # Counter({'a': 1, 'b': 2})

print(c1 + c2)      # Counter({'a': 3, 'b': 3, 'c': 1}) - addition
print(c1 - c2)      # Counter({'c': 1, 'a': 1}) - subtraction
print(c1 & c2)      # Counter({'a': 1, 'b': 1}) - intersection (min)
print(c1 | c2)      # Counter({'a': 2, 'b': 2, 'c': 1}) - union (max)

# Unary operations
+c1                 # Remove zero and negative counts
-c1                 # Keep only negative counts (flip sign)

# Useful methods
c = Counter("hello world")
print(c.elements())         # Iterator over elements (with repetition)
print(list(c.elements()))   # ['h', 'e', 'l', 'l', 'l', 'o', 'o', ...]

c.update("more letters")    # Add counts from another iterable
c.subtract("some letters")  # Subtract counts

# Total count
print(sum(c.values()))      # Total of all counts
```

### 2.4 ChainMap - Combining Multiple Mappings

```python
from collections import ChainMap

# Constructor
defaults = {'color': 'red', 'user': 'guest'}
environment = {'user': 'admin'}
command_line = {'color': 'blue'}

# Chain multiple mappings (first mapping takes precedence)
combined = ChainMap(command_line, environment, defaults)

print(combined['color'])    # 'blue' (from command_line)
print(combined['user'])     # 'admin' (from environment)
print(list(combined))       # ['color', 'user'] - all keys

# Dynamic chaining
combined.maps.append({'debug': True})   # Add new mapping
print(combined['debug'])    # True

# New child (creates new ChainMap with empty dict as first mapping)
child = combined.new_child()
child['temp'] = 'value'     # Only affects the new child mapping

# Use cases
# 1. Configuration hierarchy (command line > environment > defaults)
# 2. Scope resolution in programming languages
# 3. Template contexts in web frameworks

# Example: Configuration system
class Config:
    def __init__(self):
        self.defaults = {'debug': False, 'port': 8000, 'host': 'localhost'}
        self.environment = {}
        self.command_line = {}
        self.config = ChainMap(self.command_line, self.environment, self.defaults)
    
    def set_env(self, **kwargs):
        self.environment.update(kwargs)
    
    def set_cli(self, **kwargs):
        self.command_line.update(kwargs)
    
    def get(self, key, default=None):
        return self.config.get(key, default)

config = Config()
config.set_env(debug=True)
config.set_cli(port=3000)
print(config.get('debug'))  # True
print(config.get('port'))   # 3000
```

## Dictionary Performance and Optimization

### Time Complexity
```python
# Dictionary operations time complexity (average case)
d = {'a': 1, 'b': 2, 'c': 3}

# O(1) operations
d['a']              # Access
d['a'] = 10         # Assignment
'a' in d            # Membership test
del d['a']          # Deletion
len(d)              # Length

# O(n) operations
list(d.keys())      # Get all keys
list(d.values())    # Get all values
list(d.items())     # Get all items
d.copy()            # Shallow copy
```

### Memory Optimization
```python
import sys

# Dictionary memory usage
d1 = {'a': 1, 'b': 2, 'c': 3}
print(sys.getsizeof(d1))

# Pre-sizing dictionaries (Python doesn't have direct pre-sizing)
# But you can create from known items for better performance
keys = ['a', 'b', 'c', 'd', 'e']
values = [1, 2, 3, 4, 5]
d = dict(zip(keys, values))  # More efficient than incremental building

# Memory-efficient patterns
# Use slots for objects that will be dictionary keys
class Point:
    __slots__ = ['x', 'y']
    def __init__(self, x, y):
        self.x, self.y = x, y
    def __hash__(self):
        return hash((self.x, self.y))
    def __eq__(self, other):
        return self.x == other.x and self.y == other.y
```

### Best Practices

#### Dictionary Best Practices
1. **Use appropriate key types**: Prefer immutable types (str, int, tuple)
2. **Use get() for safe access**: `d.get(key, default)` instead of `d[key]`
3. **Use setdefault() for initialization**: `d.setdefault(key, []).append(item)`
4. **Use dictionary comprehensions**: More readable and often faster
5. **Consider defaultdict**: For cases where you need default values
6. **Use items() for iteration**: `for k, v in d.items()` is efficient

#### Performance Tips
```python
# Fast dictionary creation from keys
keys = ['a', 'b', 'c']
d = dict.fromkeys(keys, 0)      # {'a': 0, 'b': 0, 'c': 0}

# Avoid repeated dictionary lookups
# Slow:
if key in d:
    value = d[key]
    # use value

# Fast:
value = d.get(key)
if value is not None:
    # use value

# Or:
try:
    value = d[key]
    # use value
except KeyError:
    # handle missing key
```

#### Common Pitfalls
```python
# 1. Mutable default values in setdefault
d = {}
d.setdefault('list', []).append(1)      # OK
d.setdefault('list', []).append(2)      # Appends to same list
print(d)  # {'list': [1, 2]}

# 2. Modifying dictionary while iterating
d = {'a': 1, 'b': 2, 'c': 3}
# Wrong:
# for k in d:
#     if d[k] > 1:
#         del d[k]  # RuntimeError: dictionary changed size during iteration

# Correct:
keys_to_delete = [k for k, v in d.items() if v > 1]
for k in keys_to_delete:
    del d[k]

# 3. Using mutable objects as keys
# Wrong:
# d = {[1, 2]: 'value'}  # TypeError: unhashable type: 'list'

# Correct:
d = {(1, 2): 'value'}   # Tuples are hashable
```

```python
import sys
```

#### 2.4.3 Specialized Dictionary Types
```python
from collections import defaultdict, OrderedDict, ChainMap, Counter

# defaultdict - provides default values
dd = defaultdict(list)
dd['key1'].append('value1')  # No KeyError
print(dd)  # defaultdict(<class 'list'>, {'key1': ['value1']})

# OrderedDict - maintains insertion order (less relevant in Python 3.7+)
od = OrderedDict([('a', 1), ('b', 2), ('c', 3)])
od.move_to_end('a')  # Move 'a' to end
print(od)

# ChainMap - combines multiple dictionaries
dict1 = {'a': 1, 'b': 2}
dict2 = {'c': 3, 'd': 4}
cm = ChainMap(dict1, dict2)
print(cm['a'])  # 1
print(list(cm))  # ['a', 'b', 'c', 'd']

# Counter - counts hashable objects
text = "hello world"
counter = Counter(text)
print(counter.most_common(3))  # [('l', 3), ('o', 2), ('h', 1)]

# Counter arithmetic
counter1 = Counter({'a': 3, 'b': 1})
counter2 = Counter({'a': 1, 'b': 2})
print(counter1 + counter2)  # Counter({'a': 4, 'b': 3})
print(counter1 - counter2)  # Counter({'a': 2})
print(counter1 & counter2)  # Intersection: Counter({'a': 1, 'b': 1})
print(counter1 | counter2)  # Union: Counter({'a': 3, 'b': 2})
```




### 4.3 Mapping Types

#### 4.3.1 Dictionary (dict) - Hash Table Implementation
```python
# Dictionary implementation details
# - Hash table with open addressing
# - Key objects must be hashable
# - Order preserved (Python 3.7+)

# Hash function and collision handling
print(hash("hello"))                      # String hash
print(hash(42))                          # Integer hash
print(hash((1, 2, 3)))                   # Tuple hash (if all elements hashable)

# Dictionary comprehensions
squares_dict = {x: x**2 for x in range(5)}
filtered_dict = {k: v for k, v in squares_dict.items() if v > 5}

# Advanced dictionary operations
data = {'a': 1, 'b': 2, 'c': 3}

# Safe access methods
print(data.get('d', 'default'))          # 'default'
print(data.setdefault('d', 4))           # 4 (adds key if missing)

# Dictionary views (dynamic)
keys = data.keys()
values = data.values()
items = data.items()

data['e'] = 5                            # Views automatically update
print(list(keys))                        # ['a', 'b', 'c', 'd', 'e']

# Dictionary merging (Python 3.9+)
dict1 = {'a': 1, 'b': 2}
dict2 = {'b': 3, 'c': 4}
merged = dict1 | dict2                   # {'a': 1, 'b': 3, 'c': 4}

# ChainMap for layered dictionaries
from collections import ChainMap
defaults = {'color': 'red', 'user': 'guest'}
settings = {'user': 'admin'}
combined = ChainMap(settings, defaults)
print(combined['user'])                  # 'admin'
print(combined['color'])                 # 'red'
```
